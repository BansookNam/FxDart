// Shapes that let a terminal operator take a fast path
fx(list).map(f).toList()          // hands the copy to the SDK's bulk fill
fx(list).filter(p).toList()       // same
fx(list).map(f).distinct()        // fuses into one stage, one loop
fxFoldBy(key, seed, f, list)        // no per-key List, unlike groupBy + fold
fxCountBy(key, list)                // foldBy with a counter, pre-named
fxUniqStrict(list)                  // dedupe once when the result is reused
